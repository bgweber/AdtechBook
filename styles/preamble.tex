% =====================================================================
% KDP 7" x 10" black-and-white print styling (included in the PDF header)
% Keep these values in sync with _quarto.yml. Layout follows O'Reilly
% conventions: "Chapter N." labels, unnumbered sections, roman-numbered
% front matter, chapters opening on right-hand pages.
% =====================================================================
\newcommand{\BookTitle}{Machine Learning in Mobile AdTech}
\newcommand{\BookAuthor}{Ben G Weber}
\newcommand{\BookYear}{2026}

% ---------------------------------------------------------------------
% Gray shading for code blocks and formulas. KDP recommends at least a
% 10% grayscale fill in black-ink books; 12% gray (0.88) is used here.
% Formulas are boxed by filters/formula-box.lua.
% ---------------------------------------------------------------------
\usepackage[most]{tcolorbox}
\definecolor{formulabg}{gray}{0.88}
\definecolor{formularule}{gray}{0.5}
\newtcolorbox{formulabox}{enhanced, breakable=false, colback=formulabg,
  colframe=formularule, boxrule=0pt, leftrule=2pt, arc=0pt,
  left=6pt, right=6pt, top=1pt, bottom=1pt,
  before skip=8pt, after skip=8pt,
  before upper={\setlength{\abovedisplayskip}{3pt}%
    \setlength{\belowdisplayskip}{3pt}}}

% ---------------------------------------------------------------------
% Chapter and part labels: "Chapter 1." above the title. Sections are
% not numbered.
% ---------------------------------------------------------------------
\renewcommand*{\chapterformat}{\chapapp~\thechapter.\enskip}
\renewcommand*{\partformat}{\partname~\thepart.}
\setcounter{secnumdepth}{0}
% Let pages end a line or two short instead of stretching the space around
% headings to fill every page to the same depth.
\raggedbottom
% Keep cross-reference words whole (e.g. "Chapter 2" not "Chap-ter 2").
\hyphenation{Chapter Chapters Appendix Figure Table}
% Let a tall figure take up to 85% of a page, so figures stay near the
% paragraphs that introduce them instead of drifting to later pages.
\renewcommand{\topfraction}{0.85}
\renewcommand{\bottomfraction}{0.6}
\renewcommand{\textfraction}{0.1}
\renewcommand{\floatpagefraction}{0.75}
% Restart footnote numbers in every chapter, including unnumbered ones
% such as the Welcome page and the Preface.
\AddToHook{cmd/chapter/before}{\setcounter{footnote}{0}}

% Compact spacing, closer to a typical technical book: small gaps above
% headings, almost none between a heading and its text, and a small gap
% between paragraphs. (Negative beforeskip suppresses the paragraph indent.)
\RedeclareSectionCommand[
  beforeskip=-0.8\baselineskip plus -0.1\baselineskip minus -0.1\baselineskip,
  afterskip=0.15\baselineskip]{section}
\RedeclareSectionCommand[
  beforeskip=-0.6\baselineskip plus -0.1\baselineskip,
  afterskip=0.1\baselineskip]{subsection}
\RedeclareSectionCommand[
  beforeskip=-0.5\baselineskip plus -0.1\baselineskip,
  afterskip=0.1\baselineskip]{subsubsection}
% Space around floats (figures) and captions.
\setlength{\textfloatsep}{10pt plus 2pt minus 2pt}
\setlength{\floatsep}{8pt plus 2pt minus 2pt}
\setlength{\intextsep}{8pt plus 2pt minus 2pt}

% ---------------------------------------------------------------------
% Running heads (KOMA-Script): chapter title on left pages, section title
% on right pages, page number on the outside edge. Chapter openers show
% only a centered page number in the footer.
% ---------------------------------------------------------------------
\usepackage{scrlayer-scrpage}
\clearpairofpagestyles
\automark[section]{chapter}
\ohead{\pagemark}
\ihead{\headmark}
\cfoot[\pagemark]{}
\renewcommand*{\chaptermarkformat}{\chapapp~\thechapter.\enskip}
\renewcommand*{\sectionmarkformat}{}
\setkomafont{pageheadfoot}{\normalfont\small\itshape}
\setkomafont{pagenumber}{\normalfont\small}

% ---------------------------------------------------------------------
% Table of contents: room for two-digit chapter numbers ("10.")
% ---------------------------------------------------------------------
\RedeclareSectionCommand[tocnumwidth=2.2em,
  beforeskip=0pt,                % no extra space above "Chapter N."
  innerskip=0.4\baselineskip,    % small gap between "Chapter N." and the title
  afterskip=1.2\baselineskip]{chapter} % small gap before the first paragraph

% ---------------------------------------------------------------------
% Front matter: keep the Welcome page and Preface roman-numbered. Quarto
% calls \mainmatter before the first page, so it is deferred; preface.qmd
% ends with \realmainmatter, which starts arabic numbering at Part I.
% ---------------------------------------------------------------------
\let\realmainmatter\mainmatter
\renewcommand*{\mainmatter}{}

% ---------------------------------------------------------------------
% Applied at \begin{document} so it overrides Quarto's own definitions.
% ---------------------------------------------------------------------
\makeatletter
\AtBeginDocument{%
  \date{}% no build date on the print title page
  \pagestyle{scrheadings}%
  \hypersetup{hidelinks}%
  \definecolor{shadecolor}{gray}{0.88}%
  \setlength{\parskip}{3pt plus 1pt minus 1pt}% small gap between paragraphs
  % Long code lines wrap instead of running into the margin.
  \@ifpackageloaded{fvextra}{\fvset{breaklines=true,breakanywhere=true}}{}%
  % Callout boxes in black and gray instead of color.
  \definecolor{quarto-callout-color}{HTML}{000000}%
  \definecolor{quarto-callout-note-color}{HTML}{000000}%
  \definecolor{quarto-callout-tip-color}{HTML}{000000}%
  \definecolor{quarto-callout-important-color}{HTML}{000000}%
  \definecolor{quarto-callout-warning-color}{HTML}{000000}%
  \definecolor{quarto-callout-caution-color}{HTML}{000000}%
  \definecolor{quarto-callout-color-frame}{HTML}{000000}%
  \definecolor{quarto-callout-note-color-frame}{HTML}{000000}%
  \definecolor{quarto-callout-tip-color-frame}{HTML}{000000}%
  \definecolor{quarto-callout-important-color-frame}{HTML}{000000}%
  \definecolor{quarto-callout-warning-color-frame}{HTML}{000000}%
  \definecolor{quarto-callout-caution-color-frame}{HTML}{000000}%
}
\makeatother
